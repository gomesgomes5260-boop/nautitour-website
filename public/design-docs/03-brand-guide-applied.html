9Como aplicar no código

Tokens já estão em src/app/globals.css via @theme, e fontes carregadas em src/app/layout.tsx via next/font/google. Componentes existentes seguem com classes antigas (legacy aliases preservados) — migração gradual nas próximas PRs.

// src/app/layout.tsx — já aplicado
import { Fraunces, Montserrat, JetBrains_Mono } from "next/font/google";

const fraunces = Fraunces({ subsets: ["latin"], variable: "--font-display" });
const montserrat = Montserrat({ subsets: ["latin"], variable: "--font-sans" });
const mono = JetBrains_Mono({ subsets: ["latin"], variable: "--font-mono" });

/* src/app/globals.css — já aplicado */
@theme {
  --color-charcoal-700: #404040;  /* brand ★ */
  --color-red-600: #C00010;       /* brand ★ */
  --font-display: "Fraunces", serif;
  --font-sans: "Montserrat", sans-serif;
  /* ... resto dos tokens */
}
